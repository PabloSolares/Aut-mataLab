## Reglas en BNF

```
<PROGRAM> ::= <PROGRAM_Syntax>

<PROGRAM_Syntax> ::= <SYNTAX_COMPLETE>

<SYNTAX_COMPLETE> ::= <SYNTAX_COMPLETE> <PROGRAM_COMPLETE> | <PROGRAM_COMPLETE>

<PROGRAM_COMPLETE> ::= <Syntax_AP_P> | <AFD_DECLARACION> | <VER_AUTOMATAS> | <DES_AUTOMATAS> | <VAL_AUTOMATA> | error <PROGRAM_COMPLETE>

<DES_AUTOMATAS> ::= R_DESAUTOMATAS PAR_OPEN ID PAR_CLOSE SEMICOLON

<VER_AUTOMATAS> ::= R_VERAUTOMATAS PAR_OPEN PAR_CLOSE SEMICOLON

<VAL_AUTOMATA> ::= ID PAR_OPEN STRING PAR_CLOSE SEMICOLON

<Syntax_AP_P> ::= <Syntax_AP>

<AFD_DECLARACION> ::= LESS R_AFD R_NAME EQUAL STRING GREATHER <CONTENT_AFD> <AFD_CLOSE>

<Syntax_AP> ::= LESS R_AP R_NAME EQUAL STRING GREATHER <CONTENT_AP> <AP_CLOSE>

<CONTENT_AFD> ::= <LIST_CONFIG> R_TRANSICIONES COLON <LIST_TRANSATIONS>

<CONTENT_AP> ::= <LIST_CONFIG> R_TRANSICIONES COLON <LIST_TRANSATIONS_AP>

<LIST_TRANSATIONS_AP> ::= <LIST_ID_TRANSACTIONS_AP> <LIST_TRANSACTIONSP_AP>

<LIST_TRANSACTIONSP_AP> ::= <LIST_ID_TRANSACTIONS_AP> <LIST_TRANSACTIONSP_AP> | ε

<LIST_ID_TRANSACTIONS_AP> ::= <ID_TRANSACTIONS_AP>

<ID_TRANSACTIONS_AP> ::= ID PAR_OPEN <SYMBOLS> PAR_CLOSE PROD PAR_OPEN <SYMBOLS> PAR_CLOSE COMMA <SYMBOLS> COLON PAR_OPEN <SYMBOLS> PAR_CLOSE <LIST_OR_BEFORE_AP> SEMICOLON

<SYMBOLS> ::= ID | DOLLAR | HASTAG

<LIST_OR_BEFORE_AP> ::= OR <LIST_STACK_AP> <LIST_OR_BEFORE_AP> | ε

<LIST_STACK_AP> ::= <STATE_STACK_AP> <LIST_STACKP_AP>

<LIST_STACKP_AP> ::= <STATE_STACK_AP> <LIST_STACKP_AP> | ε

<STATE_STACK_AP> ::= PAR_OPEN <SYMBOLS> PAR_CLOSE PROD PAR_OPEN <SYMBOLS> PAR_CLOSE COMMA <SYMBOLS> COLON PAR_OPEN <SYMBOLS> PAR_CLOSE

<LIST_CONFIG> ::= <LIST_AFD> <LIST_CONFIGP>

<LIST_CONFIGP> ::= <LIST_AFD> <LIST_CONFIGP> | ε

<LIST_AFD> ::= <ID_ASIGN>

<ID_ASIGN> ::= ID EQUAL KEY_OPEN <LIST_IDS> KEY_CLOSE SEMICOLON

<LIST_IDS> ::= <STATE_ASSING> <STATE_ASIGNP>

<STATE_ASSING> ::= ID | INTEGER | HASTAG | DOLLAR

<STATE_ASIGNP> ::= COMMA <STATE_ASSING> <STATE_ASIGNP> | ε

<LIST_TRANSATIONS> ::= <LIST_ID_TRANSACTIONS> <LIST_TRANSACTIONSP>

<LIST_TRANSACTIONSP> ::= <LIST_ID_TRANSACTIONS> <LIST_TRANSACTIONSP> | ε

<LIST_ID_TRANSACTIONS> ::= <ID_TRANSACTIONS>

<ID_TRANSACTIONS> ::= ID PROD <LIST_FOLLOWS> <LIST_OR_BEFORE> SEMICOLON

<LIST_OR_BEFORE> ::= OR <LIST_FOLLOWS> <LIST_OR_BEFORE> | ε

<LIST_FOLLOWS> ::= <STATE_FOLLOWS> <STATE_FOLLOWSP>

<STATE_FOLLOWSP> ::= COMMA <STATE_FOLLOWS> <STATE_FOLLOWSP> | ε

<STATE_FOLLOWS> ::= ID | INTEGER

<AFD_CLOSE> ::= LESS CLOSE R_AFD GREATHER | error

<AP_CLOSE> ::= LESS CLOSE R_AP GREATHER | error
```
